cmake_minimum_required(VERSION 3.16)
project(CVIII_JP_NativeCPUHarness LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
set(JP_SRC "${CMAKE_CURRENT_LIST_DIR}/../../src")
add_library(cviii_2a03_0509 STATIC
    "${JP_SRC}/cpu_2a03.cpp"
    "${JP_SRC}/native_machine_0509.cpp"
    "${JP_SRC}/vrc6_mapper.cpp"
    "${JP_SRC}/vrc6_irq.cpp")
target_include_directories(cviii_2a03_0509 PUBLIC "${JP_SRC}")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(cviii_2a03_0509 PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_executable(cviii-headless-0509 "${CMAKE_CURRENT_LIST_DIR}/../../tools/native_probe_0509.cpp")
target_link_libraries(cviii-headless-0509 PRIVATE cviii_2a03_0509)
if(BUILD_TESTING)
    add_executable(cpu-2a03-test "${CMAKE_CURRENT_LIST_DIR}/../../tests/cpu_2a03_test.cpp")
    target_link_libraries(cpu-2a03-test PRIVATE cviii_2a03_0509)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(cpu-2a03-test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME native_cpu_2a03_0509 COMMAND cpu-2a03-test)
    add_executable(native-machine-0509-test "${CMAKE_CURRENT_LIST_DIR}/../../tests/native_machine_0509_test.cpp")
    target_link_libraries(native-machine-0509-test PRIVATE cviii_2a03_0509)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(native-machine-0509-test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME native_vrc6_cpu_bus_0509 COMMAND native-machine-0509-test)
endif()
