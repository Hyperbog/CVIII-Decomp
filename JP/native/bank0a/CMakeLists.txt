cmake_minimum_required(VERSION 3.16)
project(JPBank0AStaticRecompiler LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(jp_bank0a_ops OBJECT generated/bank0a_compiled.cpp)
target_include_directories(jp_bank0a_ops PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(jp_static_bank0a static_bank0a_driver.cpp $<TARGET_OBJECTS:jp_bank0a_ops>)
target_include_directories(jp_static_bank0a PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(jp_bank0a_ops PRIVATE /W4)
  target_compile_options(jp_bank0a_ops PRIVATE /W4)
else()
  target_compile_options(jp_static_bank0a PRIVATE -Wall -Wextra -Werror -pedantic)
  target_compile_options(jp_static_bank0a PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
enable_testing()
add_test(NAME native_bank0a_opcode_dispatch_fails_closed COMMAND jp_static_bank0a --self-test)
